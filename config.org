;; Files changed on disk: ASK before reloading, rather than silently
;; auto-reverting. When you return focus to Emacs, or switch to a buffer whose
;; file changed underneath it, offer to reload — but only if the buffer has no
;; unsaved edits (those are never touched; you still get the usual conflict
;; warning on save). Dired and other special buffers, which have no unsaved
;; state, keep auto-refreshing, since prompting there would be pointless.
(setq auto-revert-verbose nil)
(add-hook 'dired-mode-hook #'auto-revert-mode)

(defvar cs/revert-offer--pending nil
  "Non-nil while a deferred revert offer is scheduled or running.
Guards against stacking offers and against the re-entrancy that wedges the
daemon (see `cs/maybe-offer-revert').")

(defun cs/revert-offer--buffer-stale-p (buf)
  "Non-nil when BUF visits a clean, local file that changed on disk."
  (and (buffer-live-p buf)
       (buffer-local-value 'buffer-file-name buf)
       (not (buffer-modified-p buf))
       (not (file-remote-p (buffer-local-value 'buffer-file-name buf)))
       (file-exists-p (buffer-local-value 'buffer-file-name buf))
       (not (verify-visited-file-modtime buf))))

(defun cs/maybe-offer-revert (&rest _)
  "Offer to reload the selected file buffer if it changed on disk and is clean.
The actual `y-or-n-p' is DEFERRED to an idle timer rather than run inline.
This hook fires from `window-selection-change-functions' and the focus-change
function, and opening the minibuffer is itself a window-selection/focus change:
prompting synchronously here re-enters these hooks and can wedge the daemon.
Deferring (plus the `cs/revert-offer--pending' guard and an active-minibuffer
check) breaks that loop — the prompt happens once, at a safe idle moment."
  (when (and (not cs/revert-offer--pending)
             (not (active-minibuffer-window))
             (not (minibufferp)))
    (let ((buf (window-buffer (selected-window))))
      (when (cs/revert-offer--buffer-stale-p buf)
        (setq cs/revert-offer--pending t)
        (run-with-idle-timer
         0.2 nil
         (lambda ()
           (unwind-protect
               ;; Re-check: the file/buffer may have changed during the idle gap.
               (when (cs/revert-offer--buffer-stale-p buf)
                 (if (y-or-n-p (format "%s changed on disk. Reload? "
                                       (buffer-name buf)))
                     (with-current-buffer buf
                       (revert-buffer :ignore-auto :noconfirm))
                   ;; Declined — accept the on-disk modtime so we stop nagging
                   ;; until the file changes again.
                   (with-current-buffer buf (set-visited-file-modtime))))
             (setq cs/revert-offer--pending nil))))))))

(defun cs/offer-revert-on-focus ()
  "Run `cs/maybe-offer-revert' only when Emacs has just regained focus."
  (when (frame-focus-state (selected-frame))
    (cs/maybe-offer-revert)))

(add-function :after after-focus-change-function #'cs/offer-revert-on-focus)

;; C-x k kills the CURRENT buffer outright — no "which buffer?" minibuffer, no
;; "Buffer has a running process; kill it?", no "Buffer modified; kill anyway?".
;; Never asks anything. Unsaved changes are discarded (the auto-save files under
;; ~/.emacs.d/auto-save-list and the backups dir remain on disk as a safety net).
;; Scoped to the interactive key only — programmatic `kill-buffer' calls from
;; other packages keep their normal safety checks.
(defun cs/kill-this-buffer-no-prompt ()
  "Kill the current buffer with no confirmation of any kind."
  (interactive)
  (set-buffer-modified-p nil)
  (let ((kill-buffer-query-functions nil))
    (kill-buffer (current-buffer))))
(global-set-key (kbd "C-x k") #'cs/kill-this-buffer-no-prompt)

;; C-c d prompts for a path and creates it, parents and all (like `mkdir -p').
;; `C-c <letter>' is the user-reserved keyspace, so it fires in EVERY major mode
;; — unlike `C-c C-d', which go-ts/org/markdown already claim for themselves.
(defun cs/make-directory (dir)
  "Create DIR and any missing parent directories, like `mkdir -p'.
Reads a path with directory-name completion; if it does not exist
yet it is created rather than merely described."
  (interactive (list (read-directory-name "Make directory: ")))
  (make-directory dir t)
  (message "Created %s" dir))
(global-set-key (kbd "C-c d") #'cs/make-directory)

;; Stop Emacs from merely *suggesting* `M-x make-directory' when you visit or
;; save a file under a folder that doesn't exist yet — offer to create the
;; parent directory on the spot instead. Gated behind y-or-n-p, so nothing is
;; ever created silently. The visit-time hook returns nil so Emacs still opens
;; the new buffer; the save-time hook is the backstop for files moved/renamed
;; into a missing directory after the buffer already exists.
(defun cs/auto-make-parent-dirs ()
  "Offer to create the parent directory of the current file if it is missing."
  (when buffer-file-name
    (let ((dir (file-name-directory buffer-file-name)))
      (when (and dir (not (file-directory-p dir))
                 (y-or-n-p (format "Directory %s does not exist.  Create it? " dir)))
        (make-directory dir t)))))
(add-hook 'before-save-hook #'cs/auto-make-parent-dirs)
(add-hook 'find-file-not-found-functions
          (lambda () (cs/auto-make-parent-dirs) nil))
#+end_src

* Environment

=exec-path-from-shell= copies PATH/GOPATH/etc. from the login shell so
=gopls=, =dlv=, =gofumpt= are findable when launching Emacs from a desktop.
=envrc= activates =direnv= environments per-buffer.

#+begin_src emacs-lisp
(use-package exec-path-from-shell
  :init
  (setq exec-path-from-shell-arguments '("-l"))   ;; login shell only — fast
  :config
  (dolist (v '("PATH" "MANPATH" "GOPATH" "GOROOT" "GOBIN"
               "GOPROXY" "GOFLAGS" "CGO_ENABLED" "GO111MODULE"))
    (add-to-list 'exec-path-from-shell-variables v))
  (when (memq window-system '(mac ns x pgtk))
    (exec-path-from-shell-initialize))
  (unless (window-system) (exec-path-from-shell-initialize))
  ;; Make sure user bin dirs are on PATH/exec-path even if the login shell
  ;; missed them. ~/.local/bin (odin etc.) is added in .zshrc, which a
  ;; non-interactive login shell doesn't source — so exec-path-from-shell can't
  ;; see it. Add these explicitly so M-x compile and friends find the tools.
  (dolist (dir '("~/go/bin" "~/.local/bin" "~/.npm-global/bin"))
    (let ((bin (expand-file-name dir)))
      (when (file-directory-p bin)
        (add-to-list 'exec-path bin)
        (setenv "PATH" (concat bin ":" (getenv "PATH")))))))

(use-package envrc
  :hook (after-init . envrc-global-mode))
#+end_src

* GitHub Copilot — AI ghost-text completions

The Neovim side has Copilot via =copilot.lua=; this is the Emacs equivalent
([[https://github.com/copilot-emacs/copilot.el][copilot-emacs/copilot.el]]).  It is a *separate* layer from corfu: corfu is
the popup of LSP/dabbrev candidates you filter by typing, while Copilot paints
a greyed-out multi-line suggestion inline as you type.  They coexist — corfu's
popup and Copilot's ghost text can both be on screen at once.

Two one-time external steps are required before suggestions appear, neither of
which lives in this file: =M-x copilot-install-server= (downloads the Copilot
language server via npm — needs =node=, present as v26) and =M-x copilot-login=
(GitHub device-code auth in the browser).  After that, =copilot-mode= turns on
in every =prog-mode= buffer.

=TAB= accepts the suggestion.  This is safe because =copilot-completion-map= is
an *overlay* keymap that is live only while ghost text is showing — overlay maps
outrank minor-mode (yasnippet) and major-mode maps, so TAB accepts when a
suggestion is visible and does its normal job (indent / yasnippet) otherwise.
=C-<tab>= and =M-RET= (Alt-Enter, terminal-safe) are kept as alternates, and the
accept-by-word / cycle / dismiss keys round it out.

Note on corfu: if corfu's completion popup is up at the same time as ghost text,
corfu's own keymap takes TAB (the popup wins); TAB accepts Copilot when only the
ghost text is showing.

#+begin_src emacs-lisp
(use-package copilot
  :hook (prog-mode . copilot-mode)
  :custom
  (copilot-idle-delay 0.15)            ;; show ghost text shortly after you pause
  ;; Big buffers: send up to 1 MB of context (default 100 kB warned on files
  ;; like app.go or SDL headers). Beyond the limit copilot still works — it
  ;; sends a window around point — so the warning is noise; silence it.
  (copilot-max-char 1000000)
  (copilot-max-char-warning-disable t)
  ;; We use one indentation width (tab-width = 4) for every language, so when a
  ;; mode has no entry in copilot-indentation-alist, Copilot's fallback to
  ;; tab-width is already correct. Silence the "no mode-specific offset" warning.
  (copilot-indent-offset-warning-disable t)
  :bind (:map copilot-completion-map
              ("<tab>"     . copilot-accept-completion)         ;; TAB accept (GUI)
              ("TAB"       . copilot-accept-completion)         ;; TAB accept (terminal)
              ("C-<tab>"   . copilot-accept-completion)         ;; alt accept (whole)
              ("M-RET"     . copilot-accept-completion)         ;; TTY-safe accept (whole)
              ("M-<right>" . copilot-accept-completion-by-word) ;; accept one word
              ("M-n"       . copilot-next-completion)           ;; cycle suggestions
              ("M-p"       . copilot-previous-completion)
              ("C-g"       . copilot-clear-overlay)             ;; dismiss
              :map copilot-mode-map
              ("C-c TAB"   . copilot-complete)                  ;; summon a suggestion on demand
              ("C-c <tab>" . copilot-complete)))
#+end_src

=C-S-<up>= / =C-S-<down>=, then reindents it so a line dragged into a struct

* Tree-sitter

Auto-installs grammars on demand and remaps =foo-mode= to =foo-ts-mode= where
appropriate (including =go-mode= → =go-ts-mode=).

The =c= grammar ships with Arch's =tree-sitter-c= package (in =/usr/lib=, on
the loader path). The =cpp=, =json=, =yaml=, =bash=, =toml=, =dockerfile= and
=cmake= grammars are built into =~/.emacs.d/tree-sitter= via =M-x
treesit-install-language-grammar=; registering the sources here makes that
reproducible after a wipe. =treesit-auto= remaps each to its built-in
=*-ts-mode=. Schema formats without a tree-sitter mode (Protocol Buffers, Cap'n
Proto) live in =lisp/lang-formats.el= instead.

#+begin_src emacs-lisp
(with-eval-after-load 'treesit
  (dolist (src '((c          "https://github.com/tree-sitter/tree-sitter-c")
                 (cpp        "https://github.com/tree-sitter/tree-sitter-cpp")
                 (rust       "https://github.com/tree-sitter/tree-sitter-rust")
                 (html       "https://github.com/tree-sitter/tree-sitter-html")
                 (json       "https://github.com/tree-sitter/tree-sitter-json")

(defun cs/go--main-file-count (dir)
  "Count the non-test .go files in DIR that define `func main'.
One means a normal program (possibly spread across several files); more than
one means a directory of standalone single-file exercises."
  (let ((count 0))
    (dolist (f (directory-files dir t "\\.go\\'"))
      (unless (string-suffix-p "_test.go" f)
        (with-temp-buffer
          (insert-file-contents f)
          (goto-char (point-min))
          (when (re-search-forward "^func main(" nil t)
            (setq count (1+ count))))))
    count))

(defun cs/go-run-file ()
  "Save and run the current Go program in a *compilation* buffer.

Uses `go run .' (the whole package in this directory) so multi-file packages
like a real app build correctly — `go run thisfile.go' would miss the package's
other files and report them as undefined.  Falls back to `go run THISFILE' only
when the directory holds several standalone `package main' files, each its own
program (the exercise layout), where `go run .' cannot pick just one."
  (interactive)
  (unless buffer-file-name
    (user-error "Buffer is not visiting a file"))
  (when (buffer-modified-p)
    (save-buffer))
  (let ((cmd (if (> (cs/go--main-file-count default-directory) 1)
                 (concat "go run "
                         (shell-quote-argument (file-name-nondirectory buffer-file-name)))
               "go run .")))
    (compile cmd)))

* C / C++ development

Mirrors the Go setup. =clangd= is the language server (Arch ships it in the
=clang= package). It reads =compile_commands.json= or =compile_flags.txt= from
the project root — =reverie= generates the former with =make compdb=, so
go-to-definition, completion, diagnostics, and hover all work with zero extra
config. Eglot starts on file open; =M-.= then uses LSP, never =etags=.

Format-on-save runs =clangd='s formatter (which honours the project's
=.clang-format=) — but *only* when a =.clang-format= actually exists in the
tree, so we never silently impose LLVM defaults on a codebase that hasn't
opted in.

#+begin_src emacs-lisp
;; Prefer clangd with a sensible flag set. add-to-list prepends, so this entry
;; wins over eglot's built-in ("clangd" "ccls") default for these modes.
(with-eval-after-load 'eglot
  (add-to-list 'eglot-server-programs
               '((c-mode c-ts-mode c++-mode c++-ts-mode)
                 . ("clangd"
                    "--background-index"
                    "--clang-tidy"
                    "--completion-style=detailed"
                    "--header-insertion=iwyu"   ;; auto-#include the header when you accept a completion
                    "--pch-storage=memory"
                    "-j=4"))))

(defun cs/c-format-on-save ()
  "Format the buffer via clangd before save, but only when the project
defines a .clang-format — otherwise leave the source untouched."
  (when (locate-dominating-file
         (or buffer-file-name default-directory) ".clang-format")
    (add-hook 'before-save-hook #'eglot-format-buffer nil t)))

(defun cs/c-setup ()
  "Common setup for C/C++ buffers."
  (setq-local indent-tabs-mode nil          ;; .clang-format: UseTab: Never
              tab-width 4
              c-basic-offset 4
              c-ts-mode-indent-offset 4
              compile-command (if (file-exists-p "Makefile")
                                  "make "
                                "cc "))
  (subword-mode 1)
  (eglot-ensure)
  (cs/c-format-on-save))

(dolist (hook '(c-mode-hook c-ts-mode-hook c++-mode-hook c++-ts-mode-hook))
  (add-hook hook #'cs/c-setup))

(defun cs/clangd-other-file ()
  "Jump between a C/C++ source file and its header via clangd.
Smarter than name-matching — clangd uses the include graph."
  (interactive)
  (let* ((server (eglot-current-server))
         (uri (and server
                   (eglot--request server :textDocument/switchSourceHeader
                                   (eglot--TextDocumentIdentifier)))))
    (if (and (stringp uri) (not (string-empty-p uri)))
        (find-file (eglot-uri-to-path uri))
      (user-error "clangd: no matching source/header"))))

* Debugging — dape (Delve, gdb)

=dape= is a modern Debug Adapter Protocol client with no Node dependency.
Go uses =dlv= directly; C/C++/assembly use gdb's NATIVE DAP adapter
(=gdb --interpreter=dap=, built in since gdb 14).

Two gdb specifics:
- =-nx= skips =~/.gdbinit=, because that file loads gef, whose console
  context output has no place inside a DAP session.
- The =gdb= config is offered in assembly buffers too; =lisp/lang-asm.el=
  binds =C-c D= (build + debug this file) and =C-c b= (toggle breakpoint).

#+begin_src emacs-lisp
(use-package dape
  :defer t
  :custom
  (dape-buffer-window-arrangement 'right)
  (dape-inlay-hints t)
  :config
  (when-let ((cfg (alist-get 'gdb dape-configs)))
    (plist-put cfg 'command-args '("--interpreter=dap" "-nx"))
    (plist-put cfg 'modes '(c-mode c-ts-mode c++-mode c++-ts-mode
                            nasm-mode asm-mode))))
#+end_src

;; C-<backspace>: copy the ENTIRE buffer to the kill ring / system clipboard
;; without moving point. `buffer-string' reads without motion and `kill-new'
;; routes to the clipboard (select-enable-clipboard is t), so point and window
;; never move. Intentional override of `backward-kill-word', which remains on
;; M-DEL / M-<backspace>. Mirrors the LazyVim Ctrl+Backspace map. Works in GUI
;; frames; a terminal frame may not distinguish C-<backspace> from C-h.
(defun cs/copy-whole-buffer ()
  "Copy the whole buffer to the kill ring (and clipboard) without moving point."
  (interactive)
  (kill-new (buffer-string))
  (message "Buffer copied to clipboard (%d chars)" (buffer-size)))
(global-set-key (kbd "C-<backspace>") #'cs/copy-whole-buffer)
